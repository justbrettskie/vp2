```js
import { createClient } from '@supabase/supabase-js';

const url = import.meta.env.VITE_SUPABASE_URL;
const publishableKey = import.meta.env.VITE_SUPABASE_PUBLISHABLE_KEY;

  throw new Error('Missing Supabase URL or publishable key');
```

### 2. Replace `src/lib/AuthContext.jsx`

This preserves the context fields used by your `App.jsx` and protected-route components while switching authentication to Supabase Auth.

```jsx
import React, {
  createContext,
  useCallback,
  useContext,
  useEffect,
  useMemo,
  useState,
} from 'react';
import { supabase } from '@/lib/supabase';

const AuthContext = createContext(null);

function mapUser(user) {
  if (!user) return null;

  // Admin roles belong in server-managed app_metadata, never user_metadata.
  return {
    ...user,
    role: user.app_metadata?.role === 'admin' ? 'admin' : 'user',
  };
}

export function AuthProvider({ children }) {
  const [user, setUser] = useState(null);
  const [isLoadingAuth, setIsLoadingAuth] = useState(true);
  const [authChecked, setAuthChecked] = useState(false);

  const checkUserAuth = useCallback(async () => {
    setIsLoadingAuth(true);

    const { data, error } = await supabase.auth.getUser();

    setUser(error ? null : mapUser(data.user));
    setAuthChecked(true);
    setIsLoadingAuth(false);

    return { user: error ? null : data.user, error };
  }, []);

  useEffect(() => {
    let active = true;

    supabase.auth.getSession().then(({ data, error }) => {
      if (!active) return;
      setUser(error ? null : mapUser(data.session?.user));
      setAuthChecked(true);
      setIsLoadingAuth(false);
    });

    const {
      data: { subscription },
    } = supabase.auth.onAuthStateChange((_event, session) => {
      if (!active) return;
      setUser(mapUser(session?.user));
      setAuthChecked(true);
      setIsLoadingAuth(false);
    });

    return () => {
      active = false;
      subscription.unsubscribe();
    };
  }, []);

  const logout = useCallback(async (shouldRedirect = true) => {
    await supabase.auth.signOut();

    if (shouldRedirect) {
      window.location.assign('/login');
    }
  }, []);

  const navigateToLogin = useCallback(() => {
    window.location.assign('/login');
  }, []);

  const value = useMemo(
    () => ({
      user,
      isAuthenticated: Boolean(user),
      isLoadingAuth,
      authChecked,

      // Compatibility fields for the existing App.jsx:
      isLoadingPublicSettings: false,
      authError: null,
      appPublicSettings: null,

      logout,
      navigateToLogin,
      checkUserAuth,
      checkAppState: checkUserAuth,
    }),
    [user, isLoadingAuth, authChecked, logout, navigateToLogin, checkUserAuth],
  );

  return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>;
}

export function useAuth() {
  const context = useContext(AuthContext);
  if (!context) throw new Error('useAuth must be used within an AuthProvider');
  return context;
}
```

### 3. Add `src/lib/vitalpulseApi.js`

These helpers use the tables already in your project. RLS scopes reads and writes to the signed-in user.

```js
import { supabase } from '@/lib/supabase';

export async function getHealthProfile() {
  const { data, error } = await supabase
    .from('health_profiles')
    .select('*')
    .maybeSingle();

  if (error) throw error;
  return data;
}

export async function saveHealthProfile(fields) {
  const {
    data: { user },
    error: authError,
  } = await supabase.auth.getUser();

  if (authError) throw authError;
  if (!user) throw new Error('You must be signed in.');

  const { data, error } = await supabase
    .from('health_profiles')
    .upsert(
      { ...fields, user_id: user.id },
      { onConflict: 'user_id' },
    )
    .select()
    .single();

  if (error) throw error;
  return data;
}

export async function listActivitySessions({
  limit = 10,
  orderBy = 'created_at',
  ascending = false,
} = {}) {
  const { data, error } = await supabase
    .from('activity_sessions')
    .select('*')
    .order(orderBy, { ascending })
    .limit(limit);

  if (error) throw error;
  return data ?? [];
}

export async function createActivitySession(session) {
  const {
    data: { user },
    error: authError,
  } = await supabase.auth.getUser();

  if (authError) throw authError;
  if (!user) throw new Error('You must be signed in.');

  const { data, error } = await supabase
    .from('activity_sessions')
    .insert({ ...session, user_id: user.id })
    .select()
    .single();

  if (error) throw error;
  return data;
}

export async function hasPaidAccess() {
  const { data, error } = await supabase
    .from('stripe_payments')
    .select('id')
    .eq('payment_status', 'paid')
    .limit(1)
    .maybeSingle();

  if (error) throw error;
  return Boolean(data);
}
```
